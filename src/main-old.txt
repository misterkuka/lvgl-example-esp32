// ------------------ Pin Definitions ------------------
#define LCD_CS 27
#define LCD_DC 13
#define LCD_RST 4
#define LCD_BL 26
#define SPI_MOSI 23
#define SPI_SCK 18
#define SPI_MISO 19 // Not used

// ------------------ Display / SPI ------------------
lv_display_t *disp;
SPIClass lcdSPI(VSPI);

// ------------------ Software SPI Write ------------------
void spi_write_byte(uint8_t b)
{
    for (int i = 0; i < 8; i++)
    {
        digitalWrite(SPI_SCK, LOW);
        digitalWrite(SPI_MOSI, (b & 0x80) ? HIGH : LOW);
        b <<= 1;
        digitalWrite(SPI_SCK, HIGH);
    }
}

// ------------------ Direct LCD Commands ------------------
void lcd_send_command(uint8_t cmd)
{
    digitalWrite(LCD_DC, LOW);
    digitalWrite(LCD_CS, LOW);
    spi_write_byte(cmd);
    digitalWrite(LCD_CS, HIGH);
}

void lcd_fill_color(uint16_t color)
{
    lcd_send_command(0x2C); // Memory write

    digitalWrite(LCD_DC, HIGH);
    digitalWrite(LCD_CS, LOW);

    for (uint32_t i = 0; i < 320UL * 480UL; i++)
    {
        spi_write_byte(color >> 8);   // high byte
        spi_write_byte(color & 0xFF); // low byte
    }

    digitalWrite(LCD_CS, HIGH);
}

void lcd_hw_reset()
{
    pinMode(LCD_RST, OUTPUT);
    digitalWrite(LCD_RST, LOW);
    delay(20);
    digitalWrite(LCD_RST, HIGH);
    delay(150);
}

void lcd_init_pins()
{
    pinMode(LCD_CS, OUTPUT);
    pinMode(LCD_DC, OUTPUT);
    pinMode(LCD_RST, OUTPUT);
    pinMode(LCD_BL, OUTPUT);
    pinMode(SPI_MOSI, OUTPUT);
    pinMode(SPI_SCK, OUTPUT);

    digitalWrite(LCD_CS, HIGH);
    digitalWrite(LCD_DC, HIGH);
}

// ------------------ Arduino Setup ------------------

void setup()
{
    Serial.begin(9600);

    // Pins
    lcd_hw_reset();

    lcd_init_pins();

    digitalWrite(LCD_BL, HIGH); // Turn on backlight

    // SPI init (used if you want hw SPI later)
    // Minimal init sequence for ST7796
    lcd_send_command(0x11); // Sleep out
    delay(150);
    lcd_send_command(0x3A); // Pixel format
    lcd_send_command(0x55); // 16-bit RGB565
    lcd_send_command(0x29); // Display ON
    delay(50);

    // ------------------ 1️⃣ Direct color test ------------------
    lcd_fill_color(0xF800); // Red
    delay(500);
    lcd_fill_color(0x07E0); // Green
    delay(500);
    lcd_fill_color(0x001F); // Blue
    delay(500);

    // ------------------ 2️⃣ LVGL Setup ------------------
    lcdSPI.begin(SPI_SCK, SPI_MISO, SPI_MOSI, LCD_CS);

    lv_init();

    disp = lv_st7796_create(480, 320, LV_LCD_FLAG_BGR, lcd_send_cmd, lcd_send_color);
    lv_display_set_rotation(disp, LV_DISPLAY_ROTATION_0);

    // Full-screen buffer for LVGL
    size_t buf_size = 320UL * 480UL;
    lv_color_t *buf1 = (lv_color_t *)lv_malloc(buf_size * sizeof(lv_color_t));
    lv_display_set_buffers(disp, buf1, nullptr, buf_size, LV_DISPLAY_RENDER_MODE_DIRECT);

    // Display "Hello World"
    lv_obj_set_style_bg_color(lv_scr_act(), lv_color_hex(0x003a57), LV_PART_MAIN);
    lv_obj_clear_flag(lv_scr_act(), LV_OBJ_FLAG_SCROLLABLE);
    lv_obj_t *label = lv_label_create(lv_scr_act());
    lv_label_set_text(label, "Hello World");
    lv_obj_set_style_text_color(label, lv_color_white(), LV_PART_MAIN);
    lv_obj_center(label);
}

// ------------------ Arduino Loop ------------------
void loop()
{
    lv_timer_handler();
    delay(5);
}
